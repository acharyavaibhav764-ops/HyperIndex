# Version 0.5: Service Layer & Decoupling

## Objective
Decouple HTTP controllers from business logic by introducing a dedicated Service layer (`services/`), implementing FastAPI Dependency Injection (`Depends`) with `@lru_cache`, and achieving a clean, testable, enterprise-grade architecture using an **Expand-and-Contract** phased rollout:
- **v0.5.1 (Expand)**: Introduce `services/` (`IngestionService`, `RAGService`) and `api/deps.py` with FastAPI `Depends` and `@lru_cache`, refactor route handlers, and maintain full backward compatibility in `api/dependencies.py`.
- **v0.5.2 (Contract)**: Permanently remove the deprecated adapter `api/dependencies.py`, consolidate imports strictly onto `api.deps`, and complete cleanup.

---

## 1. The Problem in v0.1 - v0.4

Even with routers and schemas extracted in v0.4, route handlers still perform raw workflow orchestration directly inside HTTP functions:
- Route handler for ingestion coordinates 5 sequential steps: `upload_files` $\rightarrow$ `parse` $\rightarrow$ `split` $\rightarrow$ `embed` $\rightarrow$ `upsert`.
- Route handler for querying coordinates retrieval, prompt formatting, and Ollama invocation.
- Components (`uploader`, `retriever`, `generator`, etc.) are held as module-level global singletons in `api/dependencies.py`, causing eager execution on import and making unit testing difficult without monkey-patching globals.

---

## 2. Dedicated Service Layer (`services/`)

Extract orchestration workflows into single-responsibility service classes:

### `services/ingestion_service.py`
```python
from fastapi import UploadFile

from core import Chunker, Embeddings, Parser, Retriever, Uploader
from schemas.document import DocumentIngestItem, DocumentIngestResponse


class IngestionService:
    def __init__(
        self,
        uploader: Uploader,
        parser: Parser,
        chunker: Chunker,
        embeddings: Embeddings,
        retriever: Retriever,
    ):
        self.uploader = uploader
        self.parser = parser
        self.chunker = chunker
        self.embeddings = embeddings
        self.retriever = retriever

    def ingest_files(self, files: list[UploadFile]) -> DocumentIngestResponse:
        if not files:
            raise ValueError("At least one file is required")

        # 1. Upload files to MinIO
        uploaded_documents = self.uploader.upload_files(files)

        ingested_items = []
        for item in uploaded_documents:
            doc_id = item["document_id"]
            filename = item["filename"]
            size_bytes = item["size_bytes"]

            # 2. Download and parse text pages
            content, _ = self.uploader.download_file(doc_id)
            pages = self.parser.parse(content, filename)

            # 3. Chunk text and attach metadata
            chunks, metadatas, chunk_ids = self.chunker.split(pages, doc_id, filename)

            # 4. Generate vector embeddings
            vectors = self.embeddings.embed(chunks)

            # 5. Persist to ChromaDB
            self.retriever.upsert(
                ids=chunk_ids,
                documents=chunks,
                metadatas=metadatas,
                embeddings=vectors,
            )

            # Append to list of ingested items
            ingested_items.append(
                DocumentIngestItem(
                    document_id=doc_id,
                    filename=filename,
                    size_bytes=size_bytes,
                    collection=self.retriever.collection_name,
                    total_chunks=len(chunks),
                    status="indexed",
                )
            )

        return DocumentIngestResponse(
            message="Documents uploaded and indexed successfully",
            count=len(ingested_items),
            documents=ingested_items,
        )
```

### `services/rag_service.py`
```python
from core import Generator, Retriever
from schemas.query import QueryResponse, SourceItem


class RAGService:
    def __init__(
        self,
        retriever: Retriever,
        generator: Generator,
        retrieval_k: int = 5,
    ):
        self.retriever = retriever
        self.generator = generator
        self.retrieval_k = retrieval_k

    def answer_question(self, question: str) -> QueryResponse:
        q = question.strip() if question else ""
        if not q:
            raise ValueError("Question must not be empty")

        # 1. Retrieve relevant chunks
        retrieved = self.retriever.retrieve(q, k=self.retrieval_k)
        documents = retrieved.get("documents", [[]])[0]
        metadatas = retrieved.get("metadatas", [[]])[0]
        distances = retrieved.get("distances", [[]])[0]

        if not documents:
            raise ValueError("No indexed document chunks were found for this query")

        # 2. Build prompt and generate response with local Ollama
        prompt = self.generator.build_prompt(q, documents, metadatas)
        answer = self.generator.generate(prompt)

        # Return response
        return QueryResponse(
            question=q,
            answer=answer,
            model=self.generator.model,
            sources=[
                SourceItem(metadata=metadata, distance=distance)
                for metadata, distance in zip(metadatas, distances)
            ],
        )
```

## 3. Modern Dependency Injection (`api/deps.py`)

Manage lifecycle and injection of services cleanly via FastAPI `Depends` and `@lru_cache`:

```python
# api/deps.py
from functools import lru_cache
from fastapi import Depends

from config import settings
from core import (
    Chunker,
    Embeddings,
    Generator,
    Parser,
    Retriever,
    Uploader,
)
from services.ingestion_service import IngestionService
from services.rag_service import RAGService


# Core Component Providers (Lazy Singletons via @lru_cache)
@lru_cache
def get_uploader() -> Uploader:
    return Uploader()


@lru_cache
def get_parser() -> Parser:
    return Parser()


@lru_cache
def get_chunker() -> Chunker:
    return Chunker(chunk_size=settings.chunk_size, chunk_overlap=settings.chunk_overlap)


@lru_cache
def get_embeddings() -> Embeddings:
    return Embeddings()


@lru_cache
def get_retriever(embeddings: Embeddings = Depends(get_embeddings)) -> Retriever:
    return Retriever(embeddings=embeddings)


@lru_cache
def get_generator() -> Generator:
    return Generator(
        ollama_host=settings.ollama_host,
        ollama_port=settings.ollama_port,
        model=settings.ollama_model,
    )


# Service Providers
def get_ingestion_service(
    uploader: Uploader = Depends(get_uploader),
    parser: Parser = Depends(get_parser),
    chunker: Chunker = Depends(get_chunker),
    embeddings: Embeddings = Depends(get_embeddings),
    retriever: Retriever = Depends(get_retriever),
) -> IngestionService:
    return IngestionService(
        uploader=uploader,
        parser=parser,
        chunker=chunker,
        embeddings=embeddings,
        retriever=retriever,
    )


def get_rag_service(
    retriever: Retriever = Depends(get_retriever),
    generator: Generator = Depends(get_generator),
) -> RAGService:
    return RAGService(
        retriever=retriever,
        generator=generator,
        retrieval_k=settings.retrieval_k,
    )
```

## 4. Ultra-Clean Route Handlers

Endpoints now become lightweight HTTP controllers focused solely on HTTP concerns:

```python
# api/v1/endpoints/documents.py
@router.post(
    "",
    status_code=status.HTTP_201_CREATED,
    response_model=DocumentIngestResponse,
    summary="Upload and ingest documents",
    description=(
        "Upload one or more documents, parse text, chunk, generate embeddings, "
        "and index them in ChromaDB synchronously."
    ),
)
async def ingest_documents(
    files: list[UploadFile] = File(...),
    service: IngestionService = Depends(get_ingestion_service),
) -> DocumentIngestResponse:
    if not files:
        raise HTTPException(status_code=400, detail="At least one file is required")

    return service.ingest_files(files)
```

```python
# api/v1/endpoints/query.py
@router.post(
    "/query",
    response_model=QueryResponse,
    summary="Query RAG pipeline",
    description="Retrieve relevant ChromaDB chunks and generate an answer using Ollama.",
)
async def query_pipeline(
    request: QueryRequest,
    service: RAGService = Depends(get_rag_service),
) -> QueryResponse:
    question = request.question.strip()
    if not question:
        raise HTTPException(status_code=400, detail="Question must not be empty")

    return service.answer_question(question)
```
