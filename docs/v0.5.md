# Version 0.5: Service Layer & Decoupling

## Objective
Decouple HTTP controllers from business logic by introducing a dedicated Service layer (`services/`), implementing FastAPI Dependency Injection (`Depends`), and achieving a clean, enterprise-grade architecture.

---

## 1. The Problem in v0.1 - v0.4

Even with routers and schemas extracted, route handlers still perform raw workflow orchestration directly inside HTTP functions:
- Route handler for indexing runs 5 sequential steps: `download_file` $\rightarrow$ `parse` $\rightarrow$ `split` $\rightarrow$ `embed` $\rightarrow$ `upsert`.
- Route handler for querying coordinates retrieval, prompt formatting, and Ollama invocation.
- Components (`uploader`, `retriever`, `generator`, etc.) are held in global scope.

---

## 2. Dedicated Service Layer (`services/`)

Extract orchestration workflows into single-responsibility service classes:

### `services/ingestion_service.py`
```python
from core import Uploader, Parser, Chunker, Embeddings, Retriever
from schemas.document import DocumentIndexResponse

class IngestionService:
    def __init__(
        self,
        uploader: Uploader,
        parser: Parser,
        chunker: Chunker,
        embeddings: Embeddings,
        retriever: Retriever,
    ):
        self.uploader = uploader
        self.parser = parser
        self.chunker = chunker
        self.embeddings = embeddings
        self.retriever = retriever

    def process_and_index(self, document_id: str) -> DocumentIndexResponse:
        # 1. Download raw file from MinIO
        content, filename = self.uploader.download_file(document_id)

        # 2. Parse text pages
        pages = self.parser.parse(content, filename)

        # 3. Chunk text and attach metadata
        chunks, metadatas, chunk_ids = self.chunker.split(pages, document_id, filename)

        # 4. Generate vector embeddings
        vectors = self.embeddings.embed(chunks)

        # 5. Persist to ChromaDB
        self.retriever.upsert(
            ids=chunk_ids,
            documents=chunks,
            metadatas=metadatas,
            embeddings=vectors,
        )

        return DocumentIndexResponse(
            document_id=document_id,
            filename=filename,
            collection=self.retriever.collection_name,
            total_chunks=len(chunks),
            indexed_chunks=len(chunks),
        )
```

### `services/rag_service.py`
```python
from core import Retriever, Generator
from schemas.query import QueryResponse, SourceItem

class RAGService:
    def __init__(self, retriever: Retriever, generator: Generator, retrieval_k: int = 5):
        self.retriever = retriever
        self.generator = generator
        self.retrieval_k = retrieval_k

    def answer_question(self, question: str) -> QueryResponse:
        # 1. Retrieve top-k context chunks
        retrieved = self.retriever.retrieve(question, k=self.retrieval_k)
        documents = retrieved.get("documents", [[]])[0]
        metadatas = retrieved.get("metadatas", [[]])[0]
        distances = retrieved.get("distances", [[]])[0]

        if not documents:
            raise ValueError("No indexed document chunks were found for this query")

        # 2. Build prompt and generate response
        prompt = self.generator.build_prompt(question, documents, metadatas)
        answer = self.generator.generate(prompt)

        return QueryResponse(
            question=question,
            answer=answer,
            model=self.generator.model,
            sources=[
                SourceItem(metadata=meta, distance=dist)
                for meta, dist in zip(metadatas, distances)
            ],
        )
```

## 3. Dependency Injection (`api/deps.py`)

Manage lifecycle and injection of services cleanly via FastAPI `Depends`:

```python
# api/deps.py
from functools import lru_cache
from fastapi import Depends
from config import settings
from core import Uploader, Parser, Chunker, Embeddings, Retriever, Generator
from services.ingestion_service import IngestionService
from services.rag_service import RAGService

@lru_cache
def get_uploader() -> Uploader:
    return Uploader()

@lru_cache
def get_parser() -> Parser:
    return Parser()

@lru_cache
def get_chunker() -> Chunker:
    return Chunker(chunk_size=settings.chunk_size, chunk_overlap=settings.chunk_overlap)

@lru_cache
def get_embeddings() -> Embeddings:
    return Embeddings()

@lru_cache
def get_retriever(embeddings: Embeddings = Depends(get_embeddings)) -> Retriever:
    return Retriever(embeddings=embeddings)

@lru_cache
def get_generator() -> Generator:
    return Generator(
        ollama_host=settings.ollama_host,
        ollama_port=settings.ollama_port,
        model=settings.ollama_model,
    )

def get_ingestion_service(
    uploader: Uploader = Depends(get_uploader),
    parser: Parser = Depends(get_parser),
    chunker: Chunker = Depends(get_chunker),
    embeddings: Embeddings = Depends(get_embeddings),
    retriever: Retriever = Depends(get_retriever),
) -> IngestionService:
    return IngestionService(uploader, parser, chunker, embeddings, retriever)

def get_rag_service(
    retriever: Retriever = Depends(get_retriever),
    generator: Generator = Depends(get_generator),
) -> RAGService:
    return RAGService(retriever, generator, retrieval_k=settings.retrieval_k)
```

## 4. Ultra-Clean Route Handlers

Endpoints now become 3-line functions focused solely on HTTP concerns:

```python
# api/v1/endpoints/documents.py
@router.post("/{document_id}/index", response_model=DocumentIndexResponse)
async def index_document(
    document_id: str,
    service: IngestionService = Depends(get_ingestion_service),
) -> DocumentIndexResponse:
    return service.process_and_index(document_id)
```

```python
# api/v1/endpoints/query.py
@router.post("/query", response_model=QueryResponse)
async def query_pipeline(
    request: QueryRequest,
    service: RAGService = Depends(get_rag_service),
) -> QueryResponse:
    return service.answer_question(request.question)
```
